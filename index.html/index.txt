window.fazerLogin = function(){

const usuario = document.getElementById('usuario').value;
const senha = document.getElementById('senha').value;

if(usuario === 'admin' && senha === '1234'){

document.getElementById('loginTela').style.display = 'none';
document.getElementById('sistema').style.display = 'block';

carregarProdutos();

}else{

document.getElementById('erro').innerText = 'Usuário ou senha inválidos';

}

}

window.sair = function(){
location.reload();
}

window.adicionarProduto = async function(){

const nome = document.getElementById('nome').value;
const quantidade = parseInt(document.getElementById('quantidade').value);
const minimo = parseInt(document.getElementById('minimo').value);
const lote = document.getElementById('lote').value;
const validade = document.getElementById('validade').value;
const localizacao = document.getElementById('localizacao').value;

if(!nome || isNaN(quantidade) || isNaN(minimo)){
alert('Preencha todos os campos');
return;
}

await addDoc(collection(db, 'produtos'), {

nome,
quantidade,
minimo,
lote,
validade,
localizacao

});

alert('Produto cadastrado');

limparCampos();

carregarProdutos();

}

function limparCampos(){

document.getElementById('nome').value = '';
document.getElementById('quantidade').value = '';
document.getElementById('minimo').value = '';
document.getElementById('lote').value = '';
document.getElementById('validade').value = '';
document.getElementById('localizacao').value = '';

}

async function carregarProdutos(){

const tabela = document.getElementById('tabelaProdutos');
const select = document.getElementById('produtoSelect');

if(!tabela || !select) return;

tabela.innerHTML = '';
select.innerHTML = '';

const querySnapshot = await getDocs(collection(db, 'produtos'));

querySnapshot.forEach((produtoDoc)=>{

const produto = produtoDoc.data();
const id = produtoDoc.id;

let status = 'Normal';
let classe = 'normal';

if(produto.quantidade <= produto.minimo){
status = 'Baixo';
classe = 'baixo';
}

const hoje = new Date();
const validade = new Date(produto.validade);
const diferenca = Math.ceil((validade - hoje) / (1000 * 60 * 60 * 24));

if(diferenca <= 7){
status = 'Vencendo';
classe = 'vencendo';
}

select.innerHTML += `
<option value="${id}">
${produto.nome}
</option>
`;

tabela.innerHTML += `
<tr>
<td>${produto.nome}</td>
<td>${produto.quantidade}</td>
<td>${produto.minimo}</td>
<td>${produto.lote}</td>
<td>${produto.validade}</td>
<td>${produto.localizacao}</td>
<td>
<span class="${classe}">
${status}
</span>
</td>
</tr>
`;

});

}

window.movimentarEstoque = async function(){

const id = document.getElementById('produtoSelect').value;
const quantidade = parseInt(document.getElementById('movQuantidade').value);
const tipo = document.getElementById('tipoMov').value;

const querySnapshot = await getDocs(collection(db, 'produtos'));

querySnapshot.forEach(async(produtoDoc)=>{

if(produtoDoc.id === id){

let produto = produtoDoc.data();
let novaQuantidade = produto.quantidade;

if(tipo === 'entrada'){

novaQuantidade += quantidade;

}else{

if(produto.quantidade < quantidade){
alert('Estoque insuficiente');
return;
}

novaQuantidade -= quantidade;

}

await updateDoc(doc(db, 'produtos', id), {
quantidade: novaQuantidade
});

alert('Estoque atualizado');

carregarProdutos();

}

});

}
